#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

const int N = 8; // 棋盘大小 8x8
int totalSolutions = 0; // 记录解的总数

// 标记数组
bool colUsed[N];          // 记录列是否被占用
bool diag1Used[2 * N];    // 记录主对角线 (row - col + N) 是否被占用
bool diag2Used[2 * N];    // 记录副对角线 (row + col) 是否被占用

// 用于存储当前解，solution[i] 表示第 i 行皇后所在的列号
int solution[N];

/**
 * 打印当前找到的一个解
 */
void printSolution() {
    cout << "Solution #" << ++totalSolutions << ":" << endl;
    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            if (solution[i] == j)
                cout << "Q "; // Q 代表皇后
            else
                cout << ". "; // . 代表空位
        }
        cout << endl;
    }
    cout << "-------------------------" << endl;
}

/**
 * 深度优先搜索 (DFS) + 回溯
 * @param row 当前正在处理的行号 (0 到 7)
 */
void solveNQueens(int row) {
    // 基准情况：如果所有行都处理完了，说明找到了一个合法解
    if (row == N) {
        printSolution();
        return;
    }

    // 尝试在当前行的每一列放置皇后
    for (int col = 0; col < N; ++col) {
        // 计算对角线的索引
        // diag1: row - col 可能为负数，所以加上 N 偏移量，范围 [0, 2*N-1]
        // diag2: row + col 范围 [0, 2*N-2]
        int d1 = row - col + N;
        int d2 = row + col;

        // 检查当前位置是否安全：列、主对角线、副对角线均未被占用
        if (!colUsed[col] && !diag1Used[d1] && !diag2Used[d2]) {
            // 1. 做选择：放置皇后，标记状态
            solution[row] = col;
            colUsed[col] = true;
            diag1Used[d1] = true;
            diag2Used[d2] = true;

            // 2. 递归：进入下一行
            solveNQueens(row + 1);

            // 3. 撤销选择（回溯）：恢复状态，以便尝试当前行的下一列
            colUsed[col] = false;
            diag1Used[d1] = false;
            diag2Used[d2] = false;
        }
    }
}

int main() {
    // 初始化标记数组
    fill(begin(colUsed), end(colUsed), false);
    fill(begin(diag1Used), end(diag1Used), false);
    fill(begin(diag2Used), end(diag2Used), false);

    cout << "Start solving 8-Queens problem..." << endl;
    
    // 从第 0 行开始搜索
    solveNQueens(0);

    cout << "Total solutions found: " << totalSolutions << endl;

    return 0;
}
